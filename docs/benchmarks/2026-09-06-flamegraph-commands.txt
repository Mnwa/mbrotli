# Run from the repository root. Tools and data are local to /tmp.

python3 docs/benchmarks/2026-09-06-flamegraph-harness.py.txt

# q6-rust: run in its own output directory to preserve perf.data

mkdir -p /tmp/mbrotli-flamegraphs/q6-rust
cd /tmp/mbrotli-flamegraphs/q6-rust
